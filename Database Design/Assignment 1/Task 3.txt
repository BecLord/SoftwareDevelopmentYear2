SELECT customer.custName, carhire.carReg, cardetails.make, cardetails.model, carhire.hireDate, outlet.outletLoc
FROM carhire
JOIN customer ON carhire.custNo = customer.custNo
JOIN cardetails ON carhire.carReg = cardetails.carReg
JOIN outlet ON carhire.outletNo = outlet.outletNo
WHERE customer.custNo = 'C100';
